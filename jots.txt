-- 5. Create a view for Monthly Revenue (The Scalable Way)
CREATE OR REPLACE VIEW monthly_revenue_stats AS
SELECT 
  to_char(check_in, 'Mon') as month,
  SUM(total_amount) as revenue,
  COUNT(id) as bookings
FROM bookings
WHERE payment_status = 'paid'
GROUP BY month, extract(month from check_in)
ORDER BY extract(month from check_in);
